// The signatures the application relies on and does not define. They belong to
// the host and to two library modules, `json_file` and `loopback_http`. Bodies
// are omitted.

// Host primitives.
control json_encode(ref map<string, string> values)
    exits encoded { string text }, encode_failed { string message };
control json_decode(string text)
    exits decoded { map<string, string> values }, invalid { string message };

// Resources: a value of these types must be handed to an operation that ends it.
resource http_listener;
resource http_connection;

// json_file
pub outcome load_result {
    loaded { map<string, string> values },
    missing,
    invalid { string message },
    read_failed { string message },
    too_large
}

pub outcome replace_result {
    saved,
    encode_failed { string message },
    write_failed { string message },
    commit_uncertain { string message },
    too_large
}

pub control load(ref string path, u32 byte_limit) exits load_result;
pub control replace(ref string path, ref map<string, string> values, u32 byte_limit)
    exits replace_result;

// loopback_http, imported as `transport`
pub outcome listen_result {
    listening { http_listener listener, string authority },
    failed { string message }
}

// `cancelled` is selected when the caller's task (or the entry, after a launch
// signal) is cancelled while the operation waits.
pub outcome cancellable_accept_result {
    accepted { http_connection connection },
    idle,
    failed { string message },
    cancelled
}

pub outcome cancellable_receive_result {
    request { string method, string target, map<string, string> headers, string body },
    malformed { string message },
    headers_too_large,
    body_too_large,
    unsupported_transfer,
    unsupported_expectation,
    timed_out,
    disconnected,
    failed { string message },
    cancelled
}

pub outcome send_result {
    sent,
    failed { string message }
}

pub outcome close_result {
    closed,
    failed { string message }
}

pub outcome failure_reply_result {
    sent,
    unavailable,
    failed { string message }
}

pub control listen(u32 port) exits listen_result;

// `accept` that stops waiting when cancelled.
pub async control accept_cancellable(ref http_listener listener, u32 idle_ms)
    exits cancellable_accept_result;

// `receive` that stops reading when cancelled; the connection stays open.
pub async control receive_cancellable(ref http_connection connection)
    exits cancellable_receive_result;

pub async control send(
    ref http_connection connection,
    u32 status,
    string body,
    string media,
    string allow
) exits send_result;

// Completes a connection like `close`, but parks while draining instead of
// blocking the executor. Use it on normal paths; contingencies use `close`.
pub async control finish(http_connection connection) exits close_result;

pub control release(http_listener listener) exits released;

@contingent
pub control close(http_connection connection) exits close_result;

// Sends a fixed generic 500 only if ordinary transmission has not started.
// This is a permitted emergency operation; it never recovers the failed work.
@contingent
pub control reply_failure(ref http_connection connection) exits failure_reply_result;
